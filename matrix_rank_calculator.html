#include <stdio.h>

int main()
{
    int matrix[10][10];
    int rows, cols;
    int i, j, k;
    int rank;

    printf("Enter number of rows: ");
    scanf("%d", &rows);

    printf("Enter number of columns: ");
    scanf("%d", &cols);

    printf("Enter matrix elements:\n");

    for(i = 0; i < rows; i++)
    {
        for(j = 0; j < cols; j++)
        {
            scanf("%d", &matrix[i][j]);
        }
    }

    rank = cols;

    for(i = 0; i < rank; i++)
    {
        if(matrix[i][i] != 0)
        {
            for(j = 0; j < rows; j++)
            {
                if(j != i)
                {
                    float ratio =
                    (float)matrix[j][i] /
                    matrix[i][i];

                    for(k = 0; k < rank; k++)
                    {
                        matrix[j][k] =
                        matrix[j][k] -
                        ratio * matrix[i][k];
                    }
                }
            }
        }
        else
        {
            int reduce = 1;

            for(j = i + 1; j < rows; j++)
            {
                if(matrix[j][i] != 0)
                {
                    for(k = 0; k < rank; k++)
                    {
                        int temp =
                        matrix[i][k];

                        matrix[i][k] =
                        matrix[j][k];

                        matrix[j][k] =
                        temp;
                    }

                    reduce = 0;
                    break;
                }
            }

            if(reduce)
            {
                rank--;

                for(j = 0; j < rows; j++)
                {
                    matrix[j][i] =
                    matrix[j][rank];
                }

                i--;
            }
        }
    }

    printf("\nRank of Matrix = %d\n", rank);

    return 0;
}

Q1. What is the rank of a matrix?
The maximum number of linearly independent rows or columns.
Q2. What is a square matrix?
A matrix having equal rows and columns.
Q3. Which data structure stores the matrix?
Two-dimensional array.
Q4. What is Gaussian Elimination?
A method used to simplify a matrix using row operations.
Q5. What is the maximum rank of a matrix?
The smaller of the number of rows or columns.
